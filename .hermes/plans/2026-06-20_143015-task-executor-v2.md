# Task Executor + Notification Manager 实现方案 (v3)

> **For Hermes:** Use subagent-driven-development skill to implement this plan task-by-task.

**Goal:** 为 Utility 菜单栏 app 添加 cron 定时任务调度执行器和统一通知管理。支持 sh/js/shortcuts/py 四种执行类型，任务失败自动推送异常通知。任务状态持久化到 SQLite。

**Architecture:** 
- `TaskScheduler` — one-shot Timer 到点触发，计算最早 nextFireTime 设一次性 Timer，触发后重算下一轮
- `TaskExecutor` — 按 exec type 分发执行，并发安全（每个任务独立 Process）
- `TaskStateDB` — GRDB 读写 `task_state` 表（task_name → last_fire）
- `NotificationManager` — 统一通知接口（success/warning/error），无需配置
- 配置重载时对比 DB 与 config，剔除已移除的任务记录
- `NSWorkspace` 睡眠/唤醒通知：唤醒时取消旧 Timer → 补触发 → 重算 next

**Tech Stack:** Swift 5.9+, AppKit, GRDB (SQLite), Yams, UserNotifications

---

## 调度模型：one-shot Timer

```
scheduleNext():
  计算所有 enabled tasks 的 nextCronDate(after: now)
  取最早的 = earliestNext
  cancel 旧 Timer
  设新 one-shot Timer(fireAt: earliestNext)

Timer 到点:
  执行所有 nextFireTime ≤ now 的任务（并发 dispatch）
  执行完 → scheduleNext() 继续下一轮

需要 reschedule 的四种时机:
  1. 应用启动         → handleMissed() + scheduleNext()
  2. Timer 到点执行后  → scheduleNext()
  3. 配置重载         → handleMissed() + scheduleNext()
  4. 系统唤醒         → handleMissed() + scheduleNext()
```

---

## 设计要点

- **持久化**：通过 GRDB 的 `task_state` 表（新增 v2 migration），不引入额外文件
- **调度精度**：one-shot Timer 精确到秒，非 60s 轮询
- **并发**：每个任务执行在 `DispatchQueue.global().qos(.utility).async` 上，不阻塞调度循环
- **补触发**：启动/重载/唤醒时 `handleMissed()`：遍历所有任务，`lastFire < now && cron 应已触发` → 补执行，然后设 `lastFire = now`
- **exec 类型识别**：`{type: "sh"|"js"|"shortcuts"|"py", command: "..."}`，字符串自动推断
- **配置重载清理**：`ConfigManager.reload()` 后比对 DB 中所有 task_name，删除不在 config.tasks 中的记录
- **debugLog**：调度触发、执行结果、失败通知、唤醒/睡眠等关键路径加 `debugLog()`

```swift
// MARK: - Task

public struct TaskConfig: Codable {
    var name: String = ""
    var trigger: String = ""   // cron 表达式 或 事件名
    var exec: TaskExecConfig = .sh("")
    var enabled: Bool = true

    public init() {}

    public init(from decoder: Decoder) throws {
        let c = try decoder.container(keyedBy: CodingKeys.self)
        name    = try c.decodeIfPresent(String.self, forKey: .name) ?? ""
        trigger = try c.decodeIfPresent(String.self, forKey: .trigger) ?? ""
        enabled = try c.decodeIfPresent(Bool.self, forKey: .enabled) ?? true
        // exec 可能是字符串（自动推断）或字典（显式声明）
        if let str = try? c.decode(String.self, forKey: .exec) {
            exec = TaskExecConfig.infer(from: str)
        } else if let dict = try? c.decode(TaskExecConfig.self, forKey: .exec) {
            exec = dict
        } else {
            exec = .sh("")
        }
    }
}

public enum TaskExecConfig: Codable {
    case sh(String)
    case js(String)
    case shortcuts(String)
    case py(String)

    enum CodingKeys: String, CodingKey { case type, command }

    public init(from decoder: Decoder) throws {
        let c = try decoder.container(keyedBy: CodingKeys.self)
        let type = try c.decode(String.self, forKey: .type)
        let cmd  = try c.decode(String.self, forKey: .command)
        switch type {
        case "sh":        self = .sh(cmd)
        case "js":        self = .js(cmd)
        case "shortcuts": self = .shortcuts(cmd)
        case "py":        self = .py(cmd)
        default: throw DecodingError.dataCorruptedError(forKey: .type, in: c, debugDescription: "Unknown exec type: \(type)")
        }
    }

    public func encode(to encoder: Encoder) throws {
        var c = encoder.container(keyedBy: CodingKeys.self)
        switch self {
        case .sh(let v):        try c.encode("sh",        forKey: .type); try c.encode(v, forKey: .command)
        case .js(let v):        try c.encode("js",        forKey: .type); try c.encode(v, forKey: .command)
        case .shortcuts(let v): try c.encode("shortcuts", forKey: .type); try c.encode(v, forKey: .command)
        case .py(let v):        try c.encode("py",        forKey: .type); try c.encode(v, forKey: .command)
        }
    }

    /// 从字符串自动推断执行类型
    static func infer(from raw: String) -> TaskExecConfig {
        let t = raw.trimmingCharacters(in: .whitespacesAndNewlines)
        if t.hasSuffix(".sh") { return .sh(t) }
        if t.hasSuffix(".js") || t.contains("Application(") { return .js(t) }
        if t.hasSuffix(".py") || t.contains("def ") || t.contains("import ") { return .py(t) }
        if t.contains("\n") { return .sh(t) }  // 多行默认 shell
        return .shortcuts(t)                    // 单行默认快捷指令名
    }
}
```

### Task 3: 实现 TaskStateDB

**Objective:** GRDB 读写 `task_state` 表，供 scheduler 查询 lastFire 和更新。

**Files:**
- Create: `Sources/Task/TaskStateDB.swift`

**Step 1: 编写文件**

```swift
import Foundation
import GRDB

public final class TaskStateDB {
    public static let shared = TaskStateDB()

    private var dbQueue: DatabaseQueue? {
        AppDatabase.shared.dbQueue
    }

    private init() {}

    /// 获取任务的上次触发时间
    public func lastFire(for taskName: String) -> Date? {
        guard let db = dbQueue else { return nil }
        return try? db.read { db in
            if let ts: Double = try Double.fetchOne(db, sql: "SELECT last_fire FROM task_state WHERE task_name = ?", arguments: [taskName]) {
                return Date(timeIntervalSince1970: ts)
            }
            return nil
        }
    }

    /// 更新任务的上次触发时间为 now
    public func setLastFire(_ date: Date, for taskName: String) {
        guard let db = dbQueue else { return }
        try? db.write { db in
            try db.execute(
                sql: "INSERT OR REPLACE INTO task_state (task_name, last_fire) VALUES (?, ?)",
                arguments: [taskName, date.timeIntervalSince1970]
            )
        }
    }

    /// 删除指定任务记录（配置重载时清理）
    public func remove(taskName: String) {
        guard let db = dbQueue else { return }
        try? db.write { db in
            try db.execute(sql: "DELETE FROM task_state WHERE task_name = ?", arguments: [taskName])
        }
    }

    /// 获取所有已记录的任务名（用于配置重载清理）
    public func allTaskNames() -> [String] {
        guard let db = dbQueue else { return [] }
        return (try? db.read { db in
            try String.fetchAll(db, sql: "SELECT task_name FROM task_state")
        }) ?? []
    }
}
```

```swift
import UserNotifications

public final class NotificationManager: NSObject {
    public static let shared = NotificationManager()

    private var granted = false

    private override init() {
        super.init()
    }

    public func requestAuthorization() {
        let center = UNUserNotificationCenter.current()
        center.delegate = self
        center.requestAuthorization(options: [.alert, .sound, .badge]) { granted, _ in
            self.granted = granted
            debugLog("[Notify] Auth \(granted ? "granted" : "denied")")
        }
    }

    public func success(_ title: String, _ body: String = "") {
        post(title: title, body: body, sound: nil)
    }

    public func warning(_ title: String, _ body: String = "") {
        post(title: title, body: body, sound: .default)
    }

    public func error(_ title: String, _ body: String = "") {
        post(title: title, body: body, sound: .defaultCritical)
    }

    private func post(title: String, body: String, sound: UNNotificationSound?) {
        guard granted else { return }

        let content = UNMutableNotificationContent()
        content.title = title
        content.body = body
        content.sound = sound

        let request = UNNotificationRequest(
            identifier: UUID().uuidString,
            content: content,
            trigger: nil
        )

        UNUserNotificationCenter.current().add(request) { error in
            if let e = error { debugLog("[Notify] Post failed: \(e)") }
        }
    }
}

// MARK: - UNUserNotificationCenterDelegate

extension NotificationManager: UNUserNotificationCenterDelegate {
    public func userNotificationCenter(
        _ center: UNUserNotificationCenter,
        willPresent notification: UNNotification,
        withCompletionHandler completionHandler: @escaping (UNNotificationPresentationOptions) -> Void
    ) {
        completionHandler([.banner, .sound])
    }
}
```

### Task 5: 实现 TaskExecutor（sh/js/shortcuts/py，并发安全）

**Objective:** 按类型执行任务，每个任务独立 Process，返回 success/output/error。

**Files:**
- Create: `Sources/Task/TaskExecutor.swift`

**Step 1: 编写文件**

```swift
import Foundation

public final class TaskExecutor {
    public struct Result {
        let success: Bool
        let output: String
        let error: String
    }

    /// 同步执行（调用方负责放到后台线程）
    public static func execute(_ config: TaskExecConfig) -> Result {
        switch config {
        case .sh(let cmd):        return runShell(cmd)
        case .js(let cmd):        return runJS(cmd)
        case .shortcuts(let cmd): return runShortcuts(cmd)
        case .py(let cmd):        return runPython(cmd)
        }
    }

    // MARK: - Shell

    private static func runShell(_ cmd: String) -> Result {
        let task = Process()
        task.executableURL = URL(fileURLWithPath: "/bin/bash")
        task.arguments = ["-c", cmd]
        return runProcess(task)
    }

    // MARK: - JavaScript (JXA)

    private static func runJS(_ code: String) -> Result {
        let task = Process()
        task.executableURL = URL(fileURLWithPath: "/usr/bin/osascript")
        task.arguments = ["-l", "JavaScript", "-e", code]
        return runProcess(task)
    }

    // MARK: - Shortcuts

    private static func runShortcuts(_ name: String) -> Result {
        let task = Process()
        task.executableURL = URL(fileURLWithPath: "/usr/bin/shortcuts")
        task.arguments = ["run", name]
        return runProcess(task)
    }

    // MARK: - Python

    private static func runPython(_ code: String) -> Result {
        let task = Process()
        task.executableURL = URL(fileURLWithPath: "/usr/bin/env")
        task.arguments = ["python3", "-c", code]
        return runProcess(task)
    }

    // MARK: - Process runner

    private static func runProcess(_ task: Process) -> Result {
        let outPipe = Pipe()
        let errPipe = Pipe()
        task.standardOutput = outPipe
        task.standardError = errPipe

        do {
            try task.run()
            task.waitUntilExit()

            let outData = outPipe.fileHandleForReading.readDataToEndOfFile()
            let errData = errPipe.fileHandleForReading.readDataToEndOfFile()
            let output = String(data: outData, encoding: .utf8) ?? ""
            let error  = String(data: errData, encoding: .utf8) ?? ""

            return Result(success: task.terminationStatus == 0, output: output, error: error)
        } catch {
            return Result(success: false, output: "", error: error.localizedDescription)
        }
    }
}
```

### Task 6: 实现 TaskScheduler（one-shot Timer + cron 匹配 + 补触发 + 并发 + 唤醒监听）

**Objective:** 主调度器：计算最早 nextFireTime 设一次性 Timer，到点执行到期任务。启动/重载/唤醒时补触发。

**Files:**
- Create: `Sources/Task/TaskScheduler.swift`

**Step 1: 编写文件**

```swift
import Cocoa

public final class TaskScheduler {
    public static let shared = TaskScheduler()

    private var timer: Timer?
    private let stateDB = TaskStateDB.shared

    private init() {}

    // MARK: - Start / Stop

    public func start() {
        handleMissed()
        scheduleNext()
    }

    public func stop() {
        timer?.invalidate()
        timer = nil
    }

    /// 配置重载后由 ConfigManager 调用
    public func reschedule() {
        timer?.invalidate()
        handleMissed()
        scheduleNext()
    }

    // MARK: - Missed catch-up

    private func handleMissed() {
        let now = Date()
        let configs = ConfigManager.shared.config.tasks.filter { $0.enabled && !$0.name.isEmpty }

        for task in configs {
            guard let expr = CronExpression(task.trigger) else { continue }

            let lastFire = stateDB.lastFire(for: task.name)
            let nextFire = nextCronDate(after: lastFire ?? .distantPast, expr: expr)
            guard let next = nextFire, next <= now else { continue }

            debugLog("[Scheduler] Catch-up fire: \(task.name)")
            fireTask(task)
            stateDB.setLastFire(now, for: task.name)
        }
        debugLog("[Scheduler] Missed catch-up done")
    }

    // MARK: - Schedule next

    private func scheduleNext() {
        let now = Date()
        let configs = ConfigManager.shared.config.tasks.filter { $0.enabled && !$0.name.isEmpty }
        var earliest: Date?

        for task in configs {
            guard let expr = CronExpression(task.trigger) else { continue }
            let lastFire = stateDB.lastFire(for: task.name)
            // 下次触发：lastFire 之后的下一个 cron 匹配点
            if let next = nextCronDate(after: lastFire ?? .distantPast, expr: expr),
               next > now {
                if earliest == nil || next < earliest! {
                    earliest = next
                }
            }
        }

        timer?.invalidate()

        guard let fireAt = earliest else {
            debugLog("[Scheduler] No upcoming tasks")
            return
        }

        let interval = fireAt.timeIntervalSince(now)
        debugLog("[Scheduler] Next fire at \(fireAt) (in \(Int(interval))s)")

        timer = Timer.scheduledTimer(withTimeInterval: max(interval, 0.1), repeats: false) { [weak self] _ in
            self?.onTimerFired()
        }
    }

    // MARK: - Timer fired

    private func onTimerFired() {
        let now = Date()
        let configs = ConfigManager.shared.config.tasks.filter { $0.enabled && !$0.name.isEmpty }

        for task in configs {
            guard let expr = CronExpression(task.trigger) else { continue }
            let lastFire = stateDB.lastFire(for: task.name)
            if let next = nextCronDate(after: lastFire ?? .distantPast, expr: expr),
               next <= now {
                debugLog("[Scheduler] Fire: \(task.name)")
                fireTask(task)
                stateDB.setLastFire(now, for: task.name)
            }
        }

        scheduleNext()
    }

    // MARK: - Execute

    private func fireTask(_ task: TaskConfig) {
        DispatchQueue.global(qos: .utility).async { [task] in
            let result = TaskExecutor.execute(task.exec)

            if result.success {
                debugLog("[Scheduler] \(task.name) OK")
            } else {
                let errMsg = result.error.isEmpty ? "exit code != 0" : String(result.error.prefix(200))
                debugLog("[Scheduler] \(task.name) FAILED: \(errMsg)")
                NotificationManager.shared.error("任务失败: \(task.name)", errMsg)
            }
        }
    }

    // MARK: - NSWorkspace sleep/wake

    public func installWakeObserver() {
        let nc = NSWorkspace.shared.notificationCenter
        nc.addObserver(forName: NSWorkspace.willSleepNotification, object: nil, queue: .main) { [weak self] _ in
            debugLog("[Scheduler] System going to sleep — invalidating timer")
            self?.timer?.invalidate()
        }
        nc.addObserver(forName: NSWorkspace.didWakeNotification, object: nil, queue: .main) { [weak self] _ in
            debugLog("[Scheduler] System woke up — rescheduling")
            self?.handleMissed()
            self?.scheduleNext()
        }
    }

    // MARK: - Cron logic

    private struct CronExpression {
        let minutes:  Set<Int>
        let hours:    Set<Int>
        let days:     Set<Int>
        let months:   Set<Int>
        let weekdays: Set<Int>

        init?(_ expr: String) {
            let parts = expr.split(separator: " ").map(String.init)
            guard parts.count == 5 else { return nil }
            guard let m  = parseField(parts[0], min: 0, max: 59),
                  let h  = parseField(parts[1], min: 0, max: 23),
                  let d  = parseField(parts[2], min: 1, max: 31),
                  let mo = parseField(parts[3], min: 1, max: 12),
                  let w  = parseField(parts[4], min: 0, max: 6) else { return nil }
            minutes = m; hours = h; days = d; months = mo; weekdays = w
        }
    }

    private func nextCronDate(after date: Date, expr: CronExpression) -> Date? {
        let calendar = Calendar.current
        var comps = calendar.dateComponents([.year, .month, .day, .hour, .minute, .second], from: date)
        comps.second = 0
        guard var current = calendar.date(from: comps) else { return nil }
        current = current.addingTimeInterval(60) // 从下一分钟开始搜

        // 最多搜索 366 天
        for _ in 0..<527040 {
            let c = calendar.dateComponents([.minute, .hour, .day, .month, .weekday], from: current)
            guard let min = c.minute, let hr = c.hour, let day = c.day,
                  let mon = c.month, let wd = c.weekday else { break }
            let cronWD = wd - 1

            if expr.minutes.contains(min), expr.hours.contains(hr),
               expr.days.contains(day), expr.months.contains(mon),
               expr.weekdays.contains(cronWD) {
                return current
            }
            current = current.addingTimeInterval(60)
        }
        return nil
    }
}

// MARK: - Field parser

private func parseField(_ s: String, min: Int, max: Int) -> Set<Int>? {
    if s == "*" { return Set(min...max) }
    var result = Set<Int>()
    for part in s.split(separator: ",") {
        let p = String(part)
        if p.contains("/") {
            let sp = p.split(separator: "/")
            guard sp.count == 2, let step = Int(sp[1]), step > 0 else { return nil }
            let range = sp[0] == "*" ? min...max : (parseRange(String(sp[0]), min: min, max: max) ?? min...max)
            for i in stride(from: range.lowerBound, through: range.upperBound, by: step) {
                result.insert(i)
            }
        } else if p.contains("-") {
            guard let r = parseRange(p, min: min, max: max) else { return nil }
            result.formUnion(r)
        } else {
            guard let v = Int(p), v >= min, v <= max else { return nil }
            result.insert(v)
        }
    }
    return result.isEmpty ? nil : result
}

private func parseRange(_ s: String, min: Int, max: Int) -> ClosedRange<Int>? {
    let sp = s.split(separator: "-")
    guard sp.count == 2, let lo = Int(sp[0]), let hi = Int(sp[1]),
          lo >= min, hi <= max, lo <= hi else { return nil }
    return lo...hi
}
```

## 变更总结

| v1 → v2 | 说明 |
|---------|------|
| 移除 `NotificationConfig` | 通知无配置节点，内置始终启用 |
| `TaskStateStore` YAML → `TaskStateDB` GRDB | 持久化用已有 SQLite，新增 v2 migration |
| `exec` 类型 `appscript` → `js` | 用 JXA (`osascript -l JavaScript`) 替代 AppleScript |
| 新增 `ConfigManager.reload()` 清理逻辑 | 配置重载时自动删除 DB 中已移除任务 |
| 新增 `debugLog()` | 调度触发/执行结果/失败通知等关键路径 |
| 新增并发 | `DispatchQueue.global(qos: .utility).async` |
